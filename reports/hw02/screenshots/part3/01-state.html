 40  def validate_planner_proposal(value: Any) -> dict[str, Any]:
 41      """Validate without changing tags, trimming text, or supplying missing values."""
 42
 43      if not isinstance(value, dict) or set(value) != {"tags", "summary"}:
 44          raise ResponseContractError("Planner object must contain only tags and summary")
 45      tags = value["tags"]
 46      if not isinstance(tags, list) or len(tags) != 3:
 47          raise ResponseContractError("tags must contain exactly three strings")
 48      if any(not isinstance(tag, str) or not tag.strip() for tag in tags):
 49          raise ResponseContractError("Every tag must be a nonblank string")
 50      summary = value["summary"]
 51      if not isinstance(summary, str) or not summary.strip():
 52          raise ResponseContractError("summary must be a nonblank string")
 53      if len(summary.split()) > 25:
 54          raise ResponseContractError("summary must contain at most 25 whitespace-delimited words")
 55      return value
