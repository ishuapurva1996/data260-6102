 10  def recursion_limit(max_turns: int) -> int:
 11      """Allow initial supervisor + worker/supervisor pairs and terminal headroom."""
 12
 13      if type(max_turns) is not int or max_turns < 1:
 14          raise ValueError("max_turns must be a positive integer")
 15      return 2 * max_turns + 10

 18  def build_graph():
 19      graph = StateGraph(AgentState)
 20      graph.add_node("supervisor", supervisor_node)
 21      graph.add_node("planner", planner_node)
 22      graph.add_node("reviewer", reviewer_node)
 23      graph.add_edge(START, "supervisor")
 24      graph.add_conditional_edges(
 25          "supervisor", router_logic,
 26          {"planner": "planner", "reviewer": "reviewer", END: END},
 27      )
 28      graph.add_edge("planner", "supervisor")
 29      graph.add_edge("reviewer", "supervisor")
 30      return graph.compile()

173  def supervisor_node(state: AgentState) -> dict[str, Any]:
174      """Commit one pending worker attempt, then apply terminal precedence."""
175
176      turns = state["turn_count"] + int(state["pending_worker_turn"])
177      updates: dict[str, Any] = {"turn_count": turns, "pending_worker_turn": False}
178      if state["error"] is not None:
179          updates.update(status="error", stop_reason=state["error"]["message"])
180      elif has_current_approval(state):
181          updates.update(status="accepted", stop_reason="Current Planner proposal approved by Reviewer.")
182      elif turns >= state["max_turns"]:
183          updates.update(status="turn_limit", stop_reason="Worker turn ceiling reached without an approved current proposal.")
184      else:
185          updates.update(status="running", stop_reason=None)
186      return updates
